 12def create_database(path):
 13    """
 14    Create an initial version of the MP dataframe
 15    """
 16    extension = path.split(".")[-1]
 17
 18    if extension == "csv":
 19        print("Read:", path)
 20        df = pl.read_csv(path)
 21
 22        # Drop columns where everything is null
 23        nulls = [df[col].null_count() == df.height for col in df.columns]
 24        nulls = zip(df.columns, nulls)
 25        for column_name, null in nulls:
 26            if null:
 27                df = df.drop(column_name)
 28
 29        new_columns = list(df.columns)
 30        for column_ix, column_name in enumerate(df.columns):
 31            if "." in column_name:
 32                new_name = column_name.split(".")[0]
 33                new_columns[column_ix] = new_name
 34
 35        df = df.rename(dict(zip(df.columns, new_columns)))
 36
 37    elif extension == "txt":
 38        print("Read:", path)
 39        f = open(path)
 40        columns = ["Riksdagsledamot", "Parti", "Valkrets"]
 41
 42        rows = []
 43        lan = None
 44        for line in f:
 45            line = line.replace("\n", "")
 46            if len(line) > 2:
 47                indented = line[:4] == "    "
 48
 49                # Non-indented lines are titles, which correspond to 'län' / region
 50                if not indented:
 51                    lan = line
 52                else:
 53                    row = line.split(",")
 54                    row = [x.strip() for x in row]
 55
 56                    datapoint = []
 57                    # Add name
 58                    name = row[0]
 59                    datapoint.append(name)
 60                    # Add party
 61                    possible_parties = [s for s in row if "f." not in s]
 62                    party = possible_parties[-1]
 63                    party = re.sub(r"\(.*?\)", "", party)
 64                    party = re.sub(r"\[.*?\]", "", party).strip()
 65                    if len(party) < 4 or party.lower() != party:
 66                        datapoint.append(party)
 67                    else:
 68                        datapoint.append(None)
 69                    # Add 'län' / region
 70                    datapoint.append(lan)
 71                    rows.append(datapoint)
 72
 73        df = pl.DataFrame(rows, schema=columns, orient="row")
 74    else:
 75        print("File type not supported.", path)
 76        return None
 77
 78    # Harmonize column names
 79    new_columns = list(df.columns)
 80    for column_ix, column_name in enumerate(df.columns):
 81        if column_name in ["Ledamot", "Riksdagsledamot", "Namn"]:
 82            new_columns[column_ix] = "name"
 83        if column_name in ["Parti"]:
 84            new_columns[column_ix] = "party"
 85        if column_name in ["Yrke"]:
 86            new_columns[column_ix] = "occupation"
 87        if column_name in ["Valkrets"]:
 88            new_columns[column_ix] = "district"
 89
 90    df = df.rename(dict(zip(df.columns, new_columns)))
 91
 92    # Drop unnecessary columns
 93    retain = ["name", "party", "district", "occupation"]
 94    for column_name in df.columns:
 95        if column_name not in retain:
 96            df = df.drop(column_name)
 97
 98    # Chamber
 99    chamber = "Enkammarriksdagen"
100    potential_chamber = path.split("/")[-2]
101    if potential_chamber == "ak":
102        chamber = "Andra kammaren"
103    elif potential_chamber == "fk":
104        chamber = "Första kammaren"
105    df = df.with_columns(pl.lit(chamber).alias("chamber"))
106
107    # Year in office
108    year_str = path.split("/")[-1].split(".")[0].replace("–­­", "-")
109    if len(year_str) == 9:
110        df = df.with_columns(
111            pl.lit(int(year_str[:4])).alias("start"),
112            pl.lit(int(year_str[-4:])).alias("end"),
113        )
114    elif len(year_str) == 4:
115        # TODO: Currently using +-3 year heuristic because exact terms
116        # are unavailable.
117        if chamber == "Första kammaren":
118            df = df.with_columns(
119                pl.lit(int(year_str) - 3).alias("start"),
120                pl.lit(int(year_str) + 3).alias("end"),
121            )
122        else:
123            df = df.with_columns(
124                pl.lit(int(year_str)).alias("start"),
125                pl.lit(int(year_str)).alias("end"),
126            )
127    else:
128        print(year_str)
129
130    return df

166def add_gender(mp_db, names):
167    """
168    Based to first names, add gender to an MP dataframe.
169    """

180            gender = "woman"

185        if "-" in first_name:
186            first_name = first_name.split("-")[0]
187        if first_name in name_to_gender:

189        else:
190            row["gender"] = None

195def clean_names(mp_db):
196    """
197    Remove artefacts from MP names and specifiers
198    """
199    print("Clean names...")
200    rows = mp_db.to_dicts()
201    for row in progressbar.progressbar(rows):
202        name = row["name"]
203        split_i = name.split(" i ")
204        if name != split_i[0]:
205            name = split_i[0]
206            if len(split_i) > 1:
207                row["specifier"] = "i " + split_i[1]
208            else:
209                row["specifier"] = None
210        if "[" in name:
211            name = name.split("[")[0]
212        if "(er" in name:
213            name = name.split("(er")[0]
214        if "ersatt av" in name:
215            name = name.split("ersatt av:")[-1]
216        name = name.strip()
217        assert name != "", "names can't be empty: " + row["name"]
218        row["name"] = name
219
220    return pl.DataFrame(rows)

223def replace_party_abbreviations(mp_db, party_db):
224    """
225    Replace party abbreviations with standardized party names.

237        if type(current_party) == str:

239            row_party = current_party.strip().lower()
240            if row_party in party_dict:
241                row["party"] = party_dict[row_party]
242
243    return pl.DataFrame(rows)

246def add_id(mp_db):
247    """
248    Generate deterministic IDs for mps based on the "name", "party", "district",
249    "chamber", "start", and "end" columns of the dataframe.
250    """
251    print("Add id...")
252    columns = ["name", "party", "district", "chamber", "start", "end"]
253    print("columns used for generation:", ", ".join(columns))
254    rows = mp_db.to_dicts()
255    for row in progressbar.progressbar(rows):
256        name = unicodedata.normalize("NFD", row["name"])
257        name = name.encode("ascii", "ignore").decode("utf-8")
258        name = name.lower().replace(" ", "_")
259        name = name.replace(".", "").replace("(", "").replace(")", "").replace(":", "")
260        party = row.get("party")
261
262        pattern = [name]
263        for column in columns:
264            value = row[column]
265            if type(value) != str:
266                value = str(value)
267            pattern.append(value)
268
269        pattern = "_".join(pattern).replace(" ", "_").lower()
270
271        digest = hashlib.md5(pattern.encode("utf-8")).hexdigest()
272        row["id"] = name + "_" + digest[:6]
273
274    return pl.DataFrame(rows)

277def add_municipality(mp_db, mun_db):
278    """
279    Add home municipalities as specifiers for matched MPs from personregister
280    """
281    original_columns = list(mp_db.columns)
282    print("Add municipalicites from 'personregister'...")
283
284    def reorder_name(name):
285        s = name.split(",")
286        if len(s) == 1:
287            return name
288        else:
289            newname = s[1].strip() + " " + s[0].strip()
290            return newname
291
292    mun_db = mun_db.with_columns(
293        pl.col("name").map_elements(reorder_name, return_dtype=pl.String),
294        ((pl.col("decade") // 10) * 10).alias("decade"),
295    ).filter(pl.col("municipality").is_not_null())
296
297    outdfs = []
298    # mp_db["municipality"] = None
299
300    start_min = min(set(mp_db["start"].to_list()))
301    end_max = max(set(mp_db["end"].to_list()))
302
303    start_min = (start_min // 10) * 10
304    end_max = (end_max // 10 + 1) * 10
305    for decade in range(start_min, end_max, 10):
306        current_mun_db = mun_db.filter(pl.col("decade") == decade)
307        current_mpdb = (
308            mp_db
309                .filter((pl.col("end") >= decade))
310                .filter(pl.col("start") < decade + 10)
311            )
312        if not current_mun_db.is_empty():
313            current_mun_db = current_mun_db.select(["name", "municipality"])
314
315            mpdb_names = current_mpdb["name"].to_list()
316
317            def rname(name):
318                nameset = set(name.split())
319
320                namesplit = name.split()
321                for mpdb_name in mpdb_names:
322                    mpdbsplit = mpdb_name.split()
323                    samefirst = namesplit[0] == mpdbsplit[0]
324                    samelast = namesplit[-1] == mpdbsplit[-1]
325                    if (
326                        samefirst
327                        and samelast
328                        and abs(len(namesplit) - len(mpdbsplit)) <= 1
329                    ):
330                        return mpdb_name.strip()
331                return name.strip()
332
333            replacedname = rname("Carl Wilhelm Höglund")
334
335            print("Carl Wilhelm Höglund", "=>", replacedname, decade)
336            current_mun_db = current_mun_db.with_columns(pl.col("name").map_elements(rname, return_dtype=pl.String))
337            merged = current_mpdb.join(current_mun_db, how="left", on="name")
338            merged.write_csv("merged_" + str(decade) + ".csv")
339
340            newnames = set(current_mun_db["name"])
341            outdfs.append(merged)
342            print(
343                "Carl Wilhelm Oskar Höglund in newnames",
344                "Carl Wilhelm Oskar Höglund" in newnames,
345            )
346        else:
347            outdfs.append(current_mpdb)
348
349    mp_db = pl.concat(outdfs, how="diagonal")
350    print(mp_db)
351    mp_db = mp_db.with_columns(
352        ("i " + pl.col("municipality").str.strip_chars()).alias("municipality")
353    ).with_columns(
354        pl.col("specifier").fill_null(pl.col("municipality")).alias("specifier")
355    ).drop("municipality")
356
357    mp_db = mp_db.group_by("id", maintain_order=True).first()
358    mp_db = mp_db.select(original_columns)
359    mp_db = mp_db.sort(["start", "chamber", "name"])
360    return mp_db
