 15def classify_paragraph(paragraph, classifier, prior=np.log([0.8, 0.2])):
 16    """
 17    Classify paragraph into speeches / descriptions with provided classifier
 18    
 19    Args:
 20        paragraph (str): the text content of a paragraph
 21        classifier (dict): a dictionary that includes 'dim', 'ft' and 'model' keys for classification
 22        prior (np.array): log prior for the classes
 23    """
 24    words = paragraph.split()
 25    V = len(words)
 26    if V == 0:
 27        return prior
 28
 29    x = np.zeros((V, classifier["dim"]))
 30
 31    ft = classifier["ft"]
 32    for ix, word in enumerate(words):
 33        vec = ft.get_word_vector(word)
 34        x[ix] = vec
 35
 36    pred = classifier["model"].predict(x, batch_size=V)
 37    return np.sum(pred, axis=0) + prior

 57def detect_speaker(matched_txt, speaker_db, metadata=None):
 58    """
 59    Detect the speaker of the house
 60
 61    Args:
 62        matched_txt (str): intro text
 63        speaker_db (pl.DataFrame): dataframe containing the speaker metadata
 64        metadata (dict): metadata about the protocol. Deprecated.
 65
 66    Returns
 67        speaker_id (str): ID as a string if detected, otherwise None
 68    """
 69    lower_txt = matched_txt.lower()
 70
 71    # Second vice speaker
 72    if re.search('andre vice', lower_txt):
 73        speaker_db = speaker_db.filter(pl.col("role").str.contains('andre'))
 74        
 75    # Third vice speaker
 76    elif re.search('tredje vice', lower_txt):
 77        speaker_db = speaker_db.filter(pl.col("role").str.contains('tredje'))
 78
 79    # First vice speaker
 80    elif re.search(r'(förste)?\svice', lower_txt):
 81        speaker_db = speaker_db.filter(pl.col("role").str.contains('förste'))
 82
 83    # Speaker
 84    elif re.search(r'(herr|fru)?\s?talman', lower_txt):
 85        speaker_db = speaker_db.filter(pl.col("role") == 'talman')
 86
 87    number_of_matches = len(set(speaker_db["id"]))
 88    if number_of_matches == 1:
 89        matched_value = speaker_db["id"][0]
 90        LOGGER.debug(f"Match found: {matched_value}")
 91        return matched_value
 92    elif number_of_matches >= 2:
 93        matched_values = set(speaker_db["id"])
 94        LOGGER.debug(f"Ambiguous matches: {matched_values}")
 95    else:
 96        LOGGER.debug(f"No matches")

 99def detect_minister(matched_txt, minister_db, intro_dict):
100    """
101    Detect a minister
102
103    Args:
104        matched_txt (str): intro text
105        minister_db (pl.DataFrame): dataframe containing the minister metadata
106        intro_dict (dict): processed information about the intro text, possibly containing eg. 'gender' 
107    
108    Returns:
109        ministed_id (str): ID as a string if detected, otherwise None
110    """
111    lower_txt = matched_txt.lower()
112
113    # Filter by gender
114    if 'gender' in intro_dict:
115        gender = intro_dict["gender"]
116        minister_db = minister_db.filter(pl.col("gender") == gender)
117
118    # Filter by date
119    if 'date' in intro_dict:
120
121        minister_db = (
122            minister_db
123                .filter(pl.col("start") <= intro_dict['date'])
124                .filter(pl.col("end") >= intro_dict['date'])
125            )
126        if not minister_db.is_empty():
127            if len(set(minister_db["id"])) == 1:
128                return minister_db["id"].first()
129
130    # Match by name
131    if 'name' in intro_dict:
132        name = intro_dict["name"].lower()
133        # thage petterson
134        #print(minister_db)
135        name_matches = names_in(name, minister_db)
136        if not name_matches.is_empty():
137            if len(set(name_matches["id"])) == 1:
138                matched_value = name_matches["id"].first()
139                LOGGER.debug(f"Match by name {matched_value}")
140                return matched_value
141
142    # Match by role
143    # Catch "utrikesdepartementet"
144    if role := re.search(r'([A-Za-zÀ-ÿ]+)(?:departementet)', lower_txt):
145        r = role.group(0).replace('departementet', '')
146        role_matches = minister_db.filter(pl.col("role").str.contains(r, literal=True))
147        if not role_matches.is_empty():
148            if len(set(role_matches["id"])) == 1:
149                matched_value = role_matches["id"].first()
150                LOGGER.debug(f"Matched by role {matched_value}")
151                return matched_value
152
153    # Catch "ministern för utrikes ärendena (...)"
154    elif role := re.search(r'(?:ministern för )([A-Za-zÀ-ÿ]+)', lower_txt):
155        r = role.group(0).split()[-1]
156        role_matches = minister_db.filter(pl.col("role").str.contains(r, literal=True))
157        if not role_matches.is_empty():
158            if len(set(role_matches["id"])) == 1:
159                return role_matches["id"][0]
160
161    elif role := re.search(r'[A-Za-zÀ-ÿ]+minister', lower_txt):
162        r = role.group(0).replace('minister', '')
163        role_matches = minister_db.filter(pl.col("role").str.contains(r, literal=True))
164        if not role_matches.is_empty():
165            if len(set(role_matches["id"])) == 1:
166                return role_matches["id"][0]

168def detect_mp(intro_dict, db, party_map=None, match_fuzzily=False):
169    """
170    Match an MP in a text snippet. Returns an MP id (str) if found, otherwise None.
171
172    If multiple people are matched, defaults to returning None.
173    """
174    LOGGER.debug(f"Detect MP based on: {intro_dict}")
175
176    intro_dict["party_abbrev"] = party_map.get(intro_dict.get("party", ""), "")
177    variables = ['party_abbrev', 'specifier', 'name']
178    variables = [v for v in variables if v in list(db.columns)] # removes missing variables
179    variables = sum([list(map(list, combinations(variables, i))) for i in range(len(variables) + 1)], [])[1:]
180    if match_fuzzily:
181        matching_funs = [name_equals, name_almost_equals, names_in]
182    else:
183        matching_funs = [name_equals, names_in]
184    return match_mp(intro_dict, db, variables, matching_funs)

186def intro_to_dict(intro_text, expressions=None):
187    """
188    Convert introduction to a metadata dictionary.
189    Tries to detect 'name', 'gender', 'party', 'specified', as well as some misc things labeled as 'other'.
190
191    Args:
192        intro_text (str): introduction text
193        expressions (list): a list of regular expressions used for metadata detection.
194            Used for optimization, the algorithm works the same if not provided.
195    
196    Returns:
197        d (dict): detected metadata
198    """
199    if expressions is None:
200        expressions = load_expressions(phase="mp")
201    intro_text = intro_text.strip()
202    d = {}
203    for exp, t in expressions:
204        m = exp.search(intro_text)
205        if m is not None:
206            matched_text = m.group(0)
207
208            if t not in d:
209                d[t] = matched_text.strip()
210                intro_text = intro_text.replace(matched_text, " ")
211
212    if "name" in d:
213        s = d["name"]
214        s = s.replace('-', ' ')
215        if ", " in s:
216            s = s.split(", ")
217            s = s[1] + " " + s[0]
218        d["name"] = s
219
220    if "gender" in d:
221        d["gender"] = d["gender"].lower()
222        if d["gender"] in ["herr", "friherre"]:
223            d["gender"] = "man"
224        if d["gender"] in ["fru", "fröken"]:
225            d["gender"] = "woman"
226
227    if "specifier" in d:
228        d["specifier"] = d["specifier"].replace("i ", "")
229        d["specifier"] = d["specifier"].replace("från ", "")
230
231    return d

233def expression_dicts(pattern_db):
234    expressions = dict()
235    manual = dict()
236    for row in pattern_db.iter_rows(named=True):
237        pattern = row["pattern"]
238        exp = re.compile(pattern)
239        # Calculate digest for distringuishing patterns without ugly characters
240        pattern_digest = hashlib.md5(pattern.encode("utf-8")).hexdigest()[:16]
241        expressions[pattern_digest] = exp
242    return expressions, manual

245def join_text(text1, text2):
246    text1, text2 = list(map(lambda x: ' '.join(x.replace('\n', ' ').split()), [text1, text2]))
247    # Account for words split over textblocks with '-'
248    if text1.endswith('-'):
249        return ''.join([text1[:-1], text2])
250    else:
251        return ' '.join([text1, text2])
